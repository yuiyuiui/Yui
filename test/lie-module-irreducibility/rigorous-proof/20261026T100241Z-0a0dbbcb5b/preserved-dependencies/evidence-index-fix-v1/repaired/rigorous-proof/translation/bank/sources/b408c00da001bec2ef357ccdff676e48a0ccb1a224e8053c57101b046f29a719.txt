theorem integral_smul_forwardFourier
    (f : ℝ → ℂ) (hf : Integrable f (volume : Measure ℝ))
    (g : SchwartzMap ℝ ℂ) :
    ∫ x : ℝ, g x • FourierTransform.fourier f x =
      ∫ x : ℝ, (FourierTransform.fourier g : SchwartzMap ℝ ℂ) x • f x := by
  rw [SchwartzMap.fourier_coe]
  change
    ∫ x : ℝ, g x •
        VectorFourier.fourierIntegral Real.fourierChar volume (innerₗ ℝ) f x =
      ∫ x : ℝ,
        VectorFourier.fourierIntegral Real.fourierChar volume (innerₗ ℝ)
          (g : ℝ → ℂ) x • f x
  have hflip : (innerₗ ℝ : ℝ →ₗ[ℝ] ℝ →ₗ[ℝ] ℝ).flip = innerₗ ℝ := by
    ext
    simp
  have h := VectorFourier.integral_fourierIntegral_smul_eq_flip
      (e := Real.fourierChar) (μ := volume) (ν := volume)
      (L := innerₗ ℝ) Real.continuous_fourierChar (by fun_prop)
      hf g.integrable
  rw [hflip] at h
  simpa [smul_eq_mul, mul_comm] using h

/-- The ordinary Fourier transform of an `L¹` function is continuous. -/
