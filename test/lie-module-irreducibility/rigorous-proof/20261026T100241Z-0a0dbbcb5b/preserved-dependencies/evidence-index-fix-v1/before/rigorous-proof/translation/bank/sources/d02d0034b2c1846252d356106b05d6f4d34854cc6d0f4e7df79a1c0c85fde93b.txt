theorem exists_global_ball_maximal_theorem_P :
    ∃ C_P : Real, 0 < C_P ∧ ∀ d : Nat, 0 < d →
      ∀ f : latticeFunction d, latticeMemL2 d f →
        ballMaxL2Norm d f ≠ ⊤ ∧
          (ballMaxL2Norm d f).toReal ≤
            C_P * (latticeL2Norm d f).toReal := by
  obtain ⟨C_L, hCL, C_B, htail⟩ := external_large_radius_ball_maximal_from
  let A : Real := (C_L + 1) ^ 2
  have hA : 0 ≤ A := sq_nonneg _
  obtain ⟨alpha1, halpha1, C_G, hCG, hgrowing⟩ :=
    exists_uniform_growingAlphaMaximal hA
  have ha : 0 < smallAlphaThreshold / 2 :=
    div_pos smallAlphaThreshold_pos (by norm_num)
  have hab : smallAlphaThreshold / 2 ≤ 2 * alpha1 := by
    have hsmall := smallAlphaThreshold_le_one_div_oneTwentyEight
    nlinarith
  obtain ⟨C_C, hCC, hcritical⟩ :=
    exists_uniform_fixedCriticalMaximal_theorem_two_two ha hab
  obtain ⟨C_S, hCS, hsmall⟩ := exists_uniform_smallAlphaMaximal
  let C_P : Real := C_S + C_C + C_G + (C_B : Real) + 1
  have hCP : 0 < C_P := by
    dsimp [C_P]
    nlinarith [hCS, hCC, hCG, C_B.2]
  refine ⟨C_P, hCP, ?_⟩
  intro d hd f hf
