theorem wuWeightedL2_memLp_two {s : ℝ} {u0 : RealField}
    (hs : 0 ≤ s) (hu0 : WuWeightedL2 s u0) :
    MemLp u0 2 (volume : Measure ℝ) := by
  have hwpos : ∀ x : ℝ, 0 < wuWeight x := by
    intro x
    unfold wuWeight
    linarith [abs_nonneg x]
  have hweight_cont : Continuous (fun x : ℝ => (wuWeight x ^ s)⁻¹) := by
    apply Continuous.inv₀
    · exact (continuous_const.add continuous_abs).rpow_const
        (fun x => Or.inl (hwpos x).ne')
    · intro x
      exact (Real.rpow_pos_of_pos (hwpos x) s).ne'
  have hu0_meas : AEStronglyMeasurable u0 volume := by
    have hprod := hweight_cont.aestronglyMeasurable.mul
      hu0.aestronglyMeasurable
    apply hprod.congr
    filter_upwards [] with x
    change (wuWeight x ^ s)⁻¹ * (wuWeight x ^ s * u0 x) = u0 x
    rw [inv_mul_eq_div]
    exact mul_div_cancel_left₀ _
      (Real.rpow_pos_of_pos (hwpos x) s).ne'
  exact hu0.mono hu0_meas (ae_of_all _ fun x => by
    have hw : 1 ≤ wuWeight x ^ s := Real.one_le_rpow
      (by unfold wuWeight; linarith [abs_nonneg x]) hs
    change |u0 x| ≤ |wuWeight x ^ s * u0 x|
    rw [abs_mul,
      abs_of_nonneg (Real.rpow_nonneg (by
        unfold wuWeight
        linarith [abs_nonneg x]) s)]
    exact le_mul_of_one_le_left (abs_nonneg _) hw)

/-- Monotonicity of Wu's weighted datum scale: a stronger spatial weight
implies every weaker one. -/
